-- Automated Full Database Backup Template with Verification
DECLARE @DatabaseName VARCHAR(100) = 'ProductionDB';
DECLARE @BackupPath VARCHAR(255) = 'C:\SQLBackups\';
DECLARE @FileName VARCHAR(255);

SET @FileName = @BackupPath + @DatabaseName + '_FULL_' + 
                CONVERT(VARCHAR(20), GETDATE(), 112) + '.bak';

BACKUP DATABASE [ProductionDB] 
TO DISK = @FileName 
WITH CHECKSUM, COMPRESSION, STATS = 10;

RESTORE VERIFYONLY FROM DISK = @FileName;
